# Odoo 19 · asistente con IA sobre WhatsApp
from odoo import api, models
from odoo.exceptions import UserError

class AsistenteIA(models.Model):
    _name = "herencia.asistente.ia"
    _inherit = ["mail.thread"]

    # UBL 2.1 validado ante la DIAN
    def generar_factura(self):
        for cita in self:
            factura = self.env["account.move"].create({
                "move_type": "out_invoice",
                "partner_id": cita.cliente_id.id,
                "x_cufe": self.env["l10n_co_edi_ext"].validar_dian(cita),
            })
            self.env["mail.message"].create({
                "body": f"Factura {factura.name} lista para enviar",
            })
        return True

    # IA conversacional en tiempo real
    def responder(self, mensaje):
        return self.env["inherit.inherit.mcp"].consultar(mensaje)
